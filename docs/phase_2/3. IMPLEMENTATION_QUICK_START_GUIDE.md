#### Create Redis Service (`backend/src/services/RedisService.ts`)
```typescript
import Redis from 'ioredis';
import logger from '../config/logger';

export class RedisService {
  private static instance: Redis;

  static getInstance(): Redis {
    if (!this.instance) {
      this.instance = new Redis({
        host: process.env.REDIS_HOST || 'localhost',
        port: parseInt(process.env.REDIS_PORT || '6379'),
        retryStrategy: (times) => {
          const delay = Math.min(times * 50, 2000);
          return delay;
        },
        reconnectOnError: (err) => {
          const targetError = 'READONLY';
          if (err.message.includes(targetError)) {
            return true; // Only reconnect when the error contains "READONLY"
          }
          return false;
        },
      });

      this.instance.on('error', (err) => {
        logger.error('Redis error:', err);
      });

      this.instance.on('connect', () => {
        logger.info('✅ Redis connected');
      });
    }

    return this.instance;
  }

  static async get<T>(key: string): Promise<T | null> {
    try {
      const value = await this.getInstance().get(key);
      return value ? JSON.parse(value) : null;
    } catch (error) {
      logger.error(`Error getting key ${key}:`, error);
      return null;
    }
  }

  static async set<T>(key: string, value: T, ttlSeconds = 300): Promise<void> {
    try {
      await this.getInstance().setex(key, ttlSeconds, JSON.stringify(value));
    } catch (error) {
      logger.error(`Error setting key ${key}:`, error);
    }
  }

  static async del(key: string): Promise<void> {
    try {
      await this.getInstance().del(key);
    } catch (error) {
      logger.error(`Error deleting key ${key}:`, error);
    }
  }

  static async invalidatePattern(pattern: string): Promise<void> {
    try {
      const keys = await this.getInstance().keys(pattern);
      if (keys.length > 0) {
        await this.getInstance().del(...keys);
      }
    } catch (error) {
      logger.error(`Error invalidating pattern ${pattern}:`, error);
    }
  }
}

export default RedisService;
```

```typescript
import Queue from 'bull';
import logger from '../config/logger';

export enum JobType {
  GENERATE_REPORT = 'generate-report',
  SEND_NOTIFICATION = 'send-notification',
  PROCESS_PREDICTIONS = 'process-predictions',
  AGGREGATE_METRICS = 'aggregate-metrics',
  SYNC_OFFLINE_DATA = 'sync-offline-data',
}

export class JobQueueService {
  private static queues: Map<string, Queue.Queue> = new Map();

  static getQueue(jobType: JobType): Queue.Queue {
    if (!this.queues.has(jobType)) {
      const queue = new Queue(jobType, {
        redis: {
          host: process.env.REDIS_HOST || 'localhost',
          port: parseInt(process.env.REDIS_PORT || '6379'),
        },
        defaultJobOptions: {
          attempts: 3,
          backoff: {
            type: 'exponential',
            delay: 2000,
          },
          removeOnComplete: true,
        },
      });

      // Event handlers
      queue.on('failed', (job, err) => {
        logger.error(`Job ${jobType} failed:`, err.message);
      });

      queue.on('completed', (job) => {
        logger.info(`Job ${jobType} completed:`, job.id);
      });

      this.queues.set(jobType, queue);
    }

    return this.queues.get(jobType)!;
  }

  static async addJob(
    jobType: JobType,
    data: any,
    options?: {
      delay?: number; // milliseconds
      priority?: number;
      removeOnComplete?: boolean;
    }
  ) {
    const queue = this.getQueue(jobType);
    return queue.add(data, options);
  }

  static processQueue(jobType: JobType, handler: (job: Queue.Job) => Promise<any>) {
    const queue = this.getQueue(jobType);
    queue.process(handler);
  }
}

export default JobQueueService;
```

### Example 1: Loyalty Service (Feature 22)

Create `backend/src/services/LoyaltyService.ts`:

```typescript
import { PrismaClient } from '@prisma/client';
import Decimal from 'decimal.js';
import logger from '../config/logger';
import { RedisService } from './RedisService';

export class LoyaltyService {
  constructor(private prisma: PrismaClient = new PrismaClient()) {}

  // Configuration
  private readonly POINTS_PER_DOLLAR = 1;
  private readonly TIER_THRESHOLDS = {
    GOLD: 100,
    PLATINUM: 500,
    DIAMOND: 2000,
  };

  /**
   * Create or get loyalty account for customer
   */
  async ensureLoyaltyAccount(tenantId: string, customerId: string) {
    const cacheKey = `loyalty:${tenantId}:${customerId}`;
    
    // Check cache first
    const cached = await RedisService.get(cacheKey);
    if (cached) return cached;

    let account = await this.prisma.loyaltyAccount.findUnique({
      where: {
        customerId,
      },
      include: {
        transactions: {
          orderBy: { createdAt: 'desc' },
          take: 5,
        },
      },
    });

    if (!account) {
      account = await this.prisma.loyaltyAccount.create({
        data: {
          tenantId,
          customerId,
          pointsBalance: new Decimal(0),
        },
        include: {
          transactions: { orderBy: { createdAt: 'desc' }, take: 5 },
        },
      });
    }

    // Cache for 1 hour
    await RedisService.set(cacheKey, account, 3600);
    return account;
  }

  /**
   * Award points on purchase
   */
  async awardPointsForPurchase(
    tenantId: string,
    customerId: string,
    amount: Decimal,
    orderId: string
  ) {
    const points = amount.times(this.POINTS_PER_DOLLAR);

    const transaction = await this.prisma.$transaction(async (tx) => {
      // Ensure account exists
      const account = await tx.loyaltyAccount.findUnique({
        where: { customerId },
      });

      if (!account) {
        throw new Error(`Loyalty account not found for customer ${customerId}`);
      }

      // Update points balance
      const updatedAccount = await tx.loyaltyAccount.update({
        where: { customerId },
        data: {
          pointsBalance: account.pointsBalance.plus(points),
          lastActivityAt: new Date(),
        },
      });

      // Create transaction record
      const txRecord = await tx.loyaltyTransaction.create({
        data: {
          tenantId,
          loyaltyAccountId: account.id,
          pointsChange: points,
          reason: 'Purchase',
          orderId,
          description: `Purchase of $${amount.toFixed(2)}`,
        },
      });

      // Check for tier upgrade
      await this.checkAndUpgradeTier(tx, tenantId, account.id, updatedAccount);

      return txRecord;
    });

    // Invalidate cache
    await RedisService.del(`loyalty:${tenantId}:${customerId}`);

    logger.info(
      `💎 Awarded ${points} loyalty points to customer ${customerId} for order ${orderId}`
    );

    return transaction;
  }

  /**
   * Redeem loyalty points
   */
  async redeemPoints(
    tenantId: string,
    customerId: string,
    pointsToRedeem: Decimal,
    reason: string
  ) {
    const account = await this.prisma.loyaltyAccount.findUnique({
      where: { customerId },
    });

    if (!account) {
      throw new Error(`Loyalty account not found`);
    }

    if (account.pointsBalance.lessThan(pointsToRedeem)) {
      throw new Error(
        `Insufficient points. Available: ${account.pointsBalance}, Requested: ${pointsToRedeem}`
      );
    }

    const result = await this.prisma.$transaction(async (tx) => {
      // Deduct points
      const updated = await tx.loyaltyAccount.update({
        where: { customerId },
        data: {
          pointsBalance: account.pointsBalance.minus(pointsToRedeem),
          lastActivityAt: new Date(),
        },
      });

      // Record transaction
      const txRecord = await tx.loyaltyTransaction.create({
        data: {
          tenantId,
          loyaltyAccountId: account.id,
          pointsChange: pointsToRedeem.times(-1),
          reason: 'Redemption',
          description: reason,
        },
      });

      return { account: updated, transaction: txRecord };
    });

    await RedisService.del(`loyalty:${tenantId}:${customerId}`);
    return result;
  }

  /**
   * Check and upgrade tier if needed
   */
  private async checkAndUpgradeTier(
    tx: any,
    tenantId: string,
    accountId: string,
    account: any
  ) {
    let newTier = account.currentTier;

    if (account.pointsBalance.gte(this.TIER_THRESHOLDS.DIAMOND)) {
      newTier = 'DIAMOND';
    } else if (account.pointsBalance.gte(this.TIER_THRESHOLDS.PLATINUM)) {
      newTier = 'PLATINUM';
    } else if (account.pointsBalance.gte(this.TIER_THRESHOLDS.GOLD)) {
      newTier = 'GOLD';
    }

    if (newTier !== account.currentTier) {
      await tx.loyaltyAccount.update({
        where: { id: accountId },
        data: { currentTier: newTier },
      });

      logger.info(
        `🎉 Customer upgraded to ${newTier} tier (${account.pointsBalance} points)`
      );
    }
  }

  /**
   * Get account summary
   */
  async getAccountSummary(customerId: string) {
    const account = await this.prisma.loyaltyAccount.findUnique({
      where: { customerId },
      include: {
        transactions: { orderBy: { createdAt: 'desc' }, take: 10 },
        customer: {
          select: { name: true, phone: true, visitCount: true, lifetimeSpend: true },
        },
      },
    });

    if (!account) {
      throw new Error('Loyalty account not found');
    }

    return {
      pointsBalance: account.pointsBalance.toNumber(),
      tier: account.currentTier,
      memberSince: account.memberSinceDate,
      recentTransactions: account.transactions,
      customerInfo: account.customer,
    };
  }
}

export default new LoyaltyService();
```

### Example 2: Caching Middleware

Create `backend/src/middleware/caching.ts`:

```typescript
import { Request, Response, NextFunction } from 'express';
import RedisService from '../services/RedisService';
import logger from '../config/logger';

/**
 * Cache GET requests for specified duration
 */
export function cacheResponse(durationSeconds: number = 300) {
  return async (req: Request, res: Response, next: NextFunction) => {
    // Only cache GET requests
    if (req.method !== 'GET') {
      return next();
    }

    const { tenantId } = req.query;
    const cacheKey = `cache:${req.path}:${JSON.stringify(req.query)}`;

    try {
      // Try to get from cache
      const cached = await RedisService.get(cacheKey);
      if (cached) {
        logger.debug(`📦 Cache hit: ${req.path}`);
        return res.json(cached);
      }
    } catch (error) {
      logger.warn('Cache retrieval failed, continuing:', error);
    }

    // Intercept res.json to cache response
    const originalJson = res.json.bind(res);
    res.json = function (data) {
      // Cache the response
      RedisService.set(cacheKey, data, durationSeconds).catch((err) => {
        logger.warn('Cache write failed:', err);
      });

      return originalJson(data);
    };

    next();
  };
}

/**
 * Invalidate cache for pattern when updating data
 */
export async function invalidateCache(pattern: string) {
  await RedisService.invalidatePattern(pattern);
  logger.debug(`🗑️ Cache invalidated: ${pattern}`);
}
```

## 📊 PERFORMANCE OPTIMIZATION TIPS

### For High-Traffic Features (Pricing, Dashboard)

```typescript
// Use batch operations
const items = await this.prisma.menuItem.findMany({
  where: { tenantId },
  include: { cost: true }, // Load cost in single query
}); // BAD: N+1 queries

// Better: Use Prisma's findMany with select
const items = await this.prisma.menuItem.findMany({
  where: { tenantId },
  select: {
    id: true,
    name: true,
    price: true,
    cost: { select: { totalCost: true } },
  },
}); // Good: Single query

// Best: Use aggregations for metrics
const metrics = await this.prisma.order.aggregate({
  where: { tenantId, closedAt: { gte: startDate } },
  _sum: { total: true },
  _count: true,
}); // Fast: Single database operation
```

### Cache Keys Pattern

```typescript
// Cache frequently accessed data with appropriate TTLs
`menus:${tenantId}` // 5 min (menu changes rarely)
`orders:today:${tenantId}` // 1 min (changes frequently)
`metrics:${tenantId}:daily` // 1 hour (calculated once/day)
`customer:${customerId}` // 10 min (balance data)
```
